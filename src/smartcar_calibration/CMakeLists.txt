cmake_minimum_required(VERSION 3.0.2)
project(smartcar_calibration)
find_package(catkin REQUIRED COMPONENTS rospy sensor_msgs cv_bridge)
catkin_python_setup()
catkin_package(CATKIN_DEPENDS rospy sensor_msgs cv_bridge)
catkin_install_python(PROGRAMS scripts/calibration_web.py DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(DIRECTORY launch web DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
if(CATKIN_ENABLE_TESTING)
  catkin_add_nosetests(tests/test_calibration.py)
endif()
